#!/bin/bash
set -e
cd "$(dirname "$0")"
echo "=== Working dir: $(pwd) ==="

apt-get update -qq && apt-get install -y -qq libgtest-dev libgmock-dev 2>/dev/null || true

rm -rf build
cmake -S . -B build -G Ninja -DPNTP_WERROR=OFF -DPNTP_BUILD_TESTS=ON -DPNTP_BUILD_BENCH=OFF 2>&1
ninja -C build pntp_tests 2>&1

# Run only Phase 4 (DNS Resolver) tests
echo ""
echo "=== Phase 4: DNS Resolver Tests ==="
./build/test/pntp_tests --gtest_filter="DNSWireFormatTest.*:DNSQueryBuilderTest.*:DNSResponseParserTest.*:DNSCacheTest.*:DNSResolverConfigTest.*:DNSGoalValidationTest.*" 2>&1

echo ""
echo "=== Phase 4: Integration Tests (requires network) ==="
./build/test/pntp_tests --gtest_filter="DNSResolverIntegrationTest.*" 2>&1 || echo "[SKIP] Integration tests require network access"
